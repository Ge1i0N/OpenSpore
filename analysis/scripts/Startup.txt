#       Add a reporter that writes to the given file under the given name
#    -addDebugReporter <name:string>
#       Add a reporter that writes to debug output
#    -addDialogReporter <name:string>
#       Add a reporter that writes to a dialog box
#    -globalLevel <level>
#       Set default log level for entire system
#    [<reporter:string>]
#       Either act on the given reporter, or list the current set of reporters
#      -level <level>
#         Set the default output level on the reporter.
#      -groupLevel <group:string> <level>
#         Set the output level for the given group on the reporter. (You can specify more than one -groupLevel option.)
#      -format <format1> ...
#         Specify one or more additional display options. Choices: none timeStamp deltaTimeStamp date group level
#        -location <level>
#           Optional addition to -format: at this level or above, the location of the original log statement will be logged.
